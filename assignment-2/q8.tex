\subsection*{Solution}

\[
	D = \begin{vmatrix}
		1 & 2 & -1 \\
		3 & -1 & 2 \\
		2 & 1 & 1
	\end{vmatrix}
\]
\begin{align*}
	D &= 1\begin{vmatrix}
		-1 & 2 \\
		1 & 1
	\end{vmatrix} - 2\begin{vmatrix}
		3 & 2 \\
		2 & 1
	\end{vmatrix} - 1\begin{vmatrix}
		3 & -1 \\
		2 & 1
	\end{vmatrix} \\
	&= 1(-3) - 2(-1) - 1(5) = -6
\end{align*}

\[
	D_x = \begin{vmatrix}
		4 & 2 & -1 \\
		1 & -1 & 2 \\
		7 & 1 & 1
	\end{vmatrix}
\]

\begin{align*}
	D_x &= 4\begin{vmatrix}
		-1 & 2 \\
		1 & 1
	\end{vmatrix} - 2\begin{vmatrix}
		1 & 2 \\
		7 & 1
	\end{vmatrix} - 1\begin{vmatrix}
		1 & -1 \\
		7 & 1
	\end{vmatrix} \\
	&= 4(-3) - 2(-13) - 1(8) = 6
\end{align*}

\[
	D_y = \begin{vmatrix}
		1 & 4 & -1 \\
		3 & 1 & 2 \\
		2 & 7 & 1
	\end{vmatrix}
\]

\begin{align*}
	D_y &= 1\begin{vmatrix}
		1 & 2 \\
		7 & 1
	\end{vmatrix} - 4\begin{vmatrix}
		3 & 2 \\
		2 & 1
	\end{vmatrix} - 1\begin{vmatrix}
		3 & 1 \\
		2 & 7
	\end{vmatrix} \\
	&= 1(-13) - 4(-1) - 1(19) = -28
\end{align*}

\[
	D_z = \begin{vmatrix}
		1 & 2 & 4 \\
		3 & -1 & 1 \\
		2 & 1 & 7
	\end{vmatrix}
\]

\begin{align*}
	D_z &= 1\begin{vmatrix}
		-1 & 1 \\
		1 & 7
	\end{vmatrix} - 2\begin{vmatrix}
		3 & 1 \\
		2 & 7
	\end{vmatrix} + 4\begin{vmatrix}
		3 & -1 \\
		2 & 1
	\end{vmatrix} \\
	&= 1(-8) - 2(19) + 4(5) = -26
\end{align*}

\begin{align*}
	x &= \frac{D_x}{D} = \frac{6}{-6} = -1 \\
	y &= \frac{D_y}{D} = \frac{-28}{-6} = \frac{14}{3} \\
	z &= \frac{D_z}{D} = \frac{-26}{-6} = \frac{13}{3}
\end{align*}

Therefore: $x = -1, y = \frac{14}{3}, z = \frac{13}{3}$
