\subsection*{Solution}

The system is:
\begin{align*}
	2x + 3y &= 7 \\
	4x - y &= 1
\end{align*}

\textbf{Step 1: Find the coefficient matrix determinant}

\[
	D = \begin{vmatrix}
		2 & 3 \\
		4 & -1
	\end{vmatrix} = 2(-1) - 3(4) = -2 - 12 = -14
\]

Since $D \neq 0$, the system has a unique solution.

\textbf{Step 2: Find $D_x$ (replace first column with constants)}

\[
	D_x = \begin{vmatrix}
		7 & 3 \\
		1 & -1
	\end{vmatrix} = 7(-1) - 3(1) = -7 - 3 = -10
\]

\textbf{Step 3: Find $D_y$ (replace second column with constants)}

\[
	D_y = \begin{vmatrix}
		2 & 7 \\
		4 & 1
	\end{vmatrix} = 2(1) - 7(4) = 2 - 28 = -26
\]

\textbf{Step 4: Apply Cramer's Rule}

\begin{align*}
	x &= \frac{D_x}{D} = \frac{-10}{-14} = \frac{5}{7} \\
	y &= \frac{D_y}{D} = \frac{-26}{-14} = \frac{13}{7}
\end{align*}

\[
	\boxed{x = \frac{5}{7}, \quad y = \frac{13}{7}}
\]

\textbf{Verification:}
\begin{align*}
	2\left(\frac{5}{7}\right) + 3\left(\frac{13}{7}\right) &= \frac{10}{7} + \frac{39}{7} = \frac{49}{7} = 7 \checkmark \\
	4\left(\frac{5}{7}\right) - \frac{13}{7} &= \frac{20}{7} - \frac{13}{7} = \frac{7}{7} = 1 \checkmark
\end{align*}
